# VGTREE v1.1 Receipts Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Tool Receipt 1.0 schema validation, bounded root-contained file loading, exact-byte digest binding, compact evidence generation, CLI commands, and public examples.

**Architecture:** Detailed receipts remain immutable sidecar JSON files below an explicitly selected receipt root. A focused `vgtree.receipts` module validates receipt structure and time semantics, reads one bounded regular file, calculates its exact-byte SHA-256 digest, and converts it into the existing compact evidence shape. Existing `record-evidence` remains the only state attachment path.

**Tech Stack:** Python 3.10+, standard library pathlib/hashlib/json/datetime, jsonschema Draft 2020-12, unittest

**Spec:** `docs/superpowers/specs/2026-08-20-vgtree-v1.1-coverage-evidence-design.md`

## Global Constraints

- Complete Capability Map and Coverage Gate deliveries first.
- Existing evidence records remain valid.
- Receipt files never live inline in workflow state.
- Passing receipt evidence is bound to exact receipt bytes by SHA-256.
- Receipt loading rejects escape from the explicit root, direct links, non-files, and files larger than 4 MiB.
- Receipt content is untrusted data and never becomes a shell command.
- Receipt output never contains credentials, environment dumps, raw prompts, cookies, authorization headers, or secret-bearing command strings.
- Evidence output uses create-only writes and never overwrites an existing path.
- The documented same-user local-directory trust boundary remains explicit; this delivery does not claim hostile multi-user filesystem isolation.

~~~python
"""Tool Receipt validation, loading, and evidence binding."""

from __future__ import annotations

import json
from datetime import datetime
from importlib.resources import files
from typing import Any

from jsonschema import Draft202012Validator

from vgtree.models import ValidationIssue, ValidationReport
from vgtree.validation import FORMAT_CHECKER


RECEIPT_SCHEMA = json.loads(
    files("vgtree").joinpath("schemas", "receipt.schema.json").read_text(encoding="utf-8")
)
RECEIPT_VALIDATOR = Draft202012Validator(RECEIPT_SCHEMA, format_checker=FORMAT_CHECKER)


def validate_receipt(value: object) -> ValidationReport:
    issues = _schema_issues(value)
    if isinstance(value, dict) and not issues:
        started = _parse_time(value["started_at"])
        ended = _parse_time(value["ended_at"])
        if ended < started:
            issues.append(
                ValidationIssue(
                    "RECEIPT_TIME_REVERSED",
                    "$.ended_at",
                    "ended_at cannot precede started_at.",
                )
            )
        if value["status"] == "PASS" and not _has_bound_proof(value):
            issues.append(
                ValidationIssue(
                    "RECEIPT_PASS_PROOF_REQUIRED",
                    "$",
                    "A passing receipt requires a bound artifact or validation.",
                )
            )
    return ValidationReport(
        status="PASS" if not issues else "FAIL",
        issues=tuple(issues),
    )
~~~

`_has_bound_proof` returns true only when at least one artifact or validation has a non-empty reference or valid digest. Date parsing uses the same RFC 3339 rules as state evidence.

import hashlib
import json
import tempfile
from pathlib import Path
from unittest.mock import patch

from vgtree.models import GuardResult
from vgtree.receipts import MAX_RECEIPT_BYTES, load_receipt


class ReceiptFileTests(unittest.TestCase):
    def test_load_receipt_hashes_exact_bytes(self) -> None:
        with tempfile.TemporaryDirectory() as directory:
            root = Path(directory) / "receipts"
            root.mkdir()
            path = root / "receipt.json"
            raw = json.dumps(valid_receipt(), ensure_ascii=False).encode("utf-8")
            path.write_bytes(raw)
            loaded = load_receipt(root, path)
            self.assertIsInstance(loaded, tuple)
            value, digest, reference = loaded
            self.assertEqual(value["receipt_id"], "receipt-build-001")
            self.assertEqual(digest, "sha256:" + hashlib.sha256(raw).hexdigest())
            self.assertEqual(reference, str(path))

    def test_path_escape_is_blocked_without_reading_content(self) -> None:
        with tempfile.TemporaryDirectory() as directory:
            base = Path(directory)
            root = base / "receipts"
            root.mkdir()
            outside = base / "outside.json"
            outside.write_text("private bytes", encoding="utf-8")
            result = load_receipt(root, outside)
            self.assertIsInstance(result, GuardResult)
            self.assertEqual(result.code, "RECEIPT_PATH_UNSAFE")
            self.assertNotIn("private bytes", str(result.as_dict()))

    def test_direct_link_is_blocked(self) -> None:
        with tempfile.TemporaryDirectory() as directory:
            root = Path(directory) / "receipts"
            root.mkdir()
            candidate = root / "receipt.json"
            with patch("vgtree.receipts.Path.is_symlink", return_value=True):
                result = load_receipt(root, candidate)
            self.assertEqual(result.code, "RECEIPT_PATH_UNSAFE")

    def test_oversized_receipt_is_rejected(self) -> None:
        with tempfile.TemporaryDirectory() as directory:
            root = Path(directory) / "receipts"
            root.mkdir()
            path = root / "receipt.json"
            path.write_bytes(b"x" * (MAX_RECEIPT_BYTES + 1))
            result = load_receipt(root, path)
            self.assertEqual(result.code, "RECEIPT_TOO_LARGE")

~~~python
def receipt_to_evidence(
    receipt: dict[str, Any], reference: str, file_digest: str
) -> GuardResult:
    report = validate_receipt(receipt)
    if not report.valid:
        return GuardResult(
            "FAIL",
            "RECEIPT_INVALID",
            "Receipt failed validation.",
            {"validation": report.as_dict()},
        )
    if not SHA256_PATTERN.fullmatch(file_digest):
        return GuardResult("FAIL", "RECEIPT_DIGEST_INVALID", "Receipt digest is invalid.")
    if not reference.strip():
        return GuardResult("FAIL", "RECEIPT_REFERENCE_REQUIRED", "Receipt reference is required.")
    evidence = {
        "id": receipt["receipt_id"],
        "type": receipt["evidence_type"],
        "subject": receipt["subject"],
        "method": receipt["evidence_method"],
        "timestamp": receipt["ended_at"],
        "outcome": receipt["status"],
        "digest": file_digest,
        "reference": reference,
    }
    evidence_report = validate_evidence(evidence)
    if not evidence_report.valid:
        return GuardResult(
            "FAIL",
            "RECEIPT_EVIDENCE_INVALID",
            "Receipt could not produce valid compact evidence.",
            {"validation": evidence_report.as_dict()},
        )
    return GuardResult(
        "PASS",
        "RECEIPT_EVIDENCE_CREATED",
        "Receipt was bound to compact evidence.",
        {"evidence": evidence},
    )
~~~
